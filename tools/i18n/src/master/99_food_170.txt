# Comida in 1.7.0: changing a food's values by hand when logging it, and a food logged before starting as that time.
# The rest of the food log is in 99_food.txt and 99_food_160.txt.

# The values card of the confirm page: "Edit values" turns its lines into fields, typed for the amount shown;
# "Restore" brings back the food's own values; the line shown while the values are ones typed by hand;
# under a field, the value saved when the one typed is less than the changed ingredients alone add ("Saved as 2 g: …")
foodlog_values_edit = Edit values ||| Editar valores
foodlog_values_follow = For the amount shown. If you change the amount or the ingredients, the values follow. ||| Para la cantidad indicada. Si cambias la cantidad o los ingredientes, los valores se recalculan.
foodlog_values_restore = Restore ||| Restaurar
foodlog_values_edited = Values changed by you ||| Valores cambiados por ti
foodlog_values_kept = Saved as %1$s: the changed ingredients alone add more ||| Se guarda %1$s: los ingredientes cambiados ya suman más

# A food logged before starts as its last entry: "As last time (Mon 2 Nov)", then what it took from it
# ("Same as then: amount, your values and ingredients", the three words joined as a list; the date has its year when it
# is another year's), and a button that starts as if it was never logged (amount, values and ingredients as usual)
foodlog_recall_title = As last time (%1$s) ||| Como la última vez (%1$s)
foodlog_recall_text = Same as then: %1$s ||| Igual que entonces: %1$s
foodlog_recall_amount = amount ||| cantidad
foodlog_recall_values = your values ||| tus valores
foodlog_recall_ingredients = ingredients ||| ingredientes
foodlog_recall_reset = Ignore last time ||| Ignorar la última vez

# Button next to the photo's grams suggestion when another amount shows ("≈ 180 g (estimated)  Use")
foodlog_estimated_use = Use ||| Usar

# A product found by its barcode that was logged before: how much last time ("Last time: 2 × Serving · 60 g")
foodlog_barcode_last_time = Last time: %1$s ||| La última vez: %1$s
