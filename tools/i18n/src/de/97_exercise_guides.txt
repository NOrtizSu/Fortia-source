# Maschinen
exg_machine_chest_press_steps = 1. Stell den Sitz so ein, dass die Griffe auf Höhe der Brustmitte sind.\n2. Setz dich mit Rücken und Kopf ans Polster, Füße flach auf dem Boden, und greif die Griffe.\n3. Drück die Griffe nach vorne, bis deine Arme fast gestreckt sind.\n4. Führ sie langsam zurück, bis du eine Dehnung in der Brust spürst, ohne dass die Gewichte aufsetzen.
exg_machine_chest_press_tips = • Halte die Schulterblätter zurück und unten am Polster.\n• Drück die Ellbogen nicht hart durch und lass die Schultern am Ende nicht nach vorne rollen.
exg_machine_incline_chest_press_steps = 1. Stell den Sitz so ein, dass die Griffe auf Höhe der oberen Brust sind.\n2. Setz dich mit dem Rücken flach an das schräge Polster und greif die Griffe.\n3. Drück nach oben und vorne, bis deine Arme fast gestreckt sind.\n4. Senk langsam ab, bis deine Hände auf Höhe der oberen Brust sind.
exg_machine_incline_chest_press_tips = • Halte die Schulterblätter fest zurück; zieh die Schultern nicht zu den Ohren hoch.\n• Halte die Ellbogen etwas unter Schulterhöhe, um die Schultern zu schonen.
exg_pec_deck_steps = 1. Stell den Sitz so ein, dass die Griffe auf Brusthöhe sind und deine Oberarme parallel zum Boden.\n2. Setz dich mit dem Rücken ans Polster und greif die Griffe (oder leg die Unterarme an die Polster), Ellbogen leicht gebeugt.\n3. Führ die Griffe in einem weiten Bogen vor der Brust zusammen.\n4. Spann kurz an, dann öffne langsam, bis du eine Dehnung über die Brust spürst.
exg_pec_deck_tips = • Halte die leichte Beugung in den Ellbogen die ganze Zeit gleich: eine Umarmung, kein Drücken.\n• Führ die Arme nicht hinter den Körper; stopp bei einer angenehmen Dehnung.
exg_machine_shoulder_press_steps = 1. Stell den Sitz so ein, dass die Griffe etwa auf Schulterhöhe beginnen.\n2. Setz dich mit dem Rücken ans Polster, Füße flach, und greif die Griffe.\n3. Drück nach oben, bis deine Arme über dem Kopf fast gestreckt sind.\n4. Senk kontrolliert zurück auf Schulterhöhe.
exg_machine_shoulder_press_tips = • Halte den unteren Rücken am Polster; mach kein Hohlkreuz, um mehr Gewicht zu drücken.\n• Drück die Ellbogen oben nicht hart durch und zieh die Schultern nicht hoch.
exg_machine_lateral_raise_steps = 1. Stell den Sitz so ein, dass deine Schultern auf Höhe der Drehpunkte der Maschine sind.\n2. Setz dich aufrecht und leg Unterarme oder Ellbogen an die Polster, Arme seitlich am Körper.\n3. Heb die Arme seitlich an, bis die Ellbogen auf Schulterhöhe sind.\n4. Senk sie langsam zum Start, ohne das Gewicht abzulegen.
exg_machine_lateral_raise_tips = • Führ die Bewegung mit den Ellbogen und halte die Schultern unten, weg von den Ohren.\n• Lehn dich nicht zurück und schwing nicht; nimm ein Gewicht, das du kontrollierst.
exg_reverse_pec_deck_steps = 1. Stell die Griffe auf die vordere Position und den Sitz so, dass sie auf Schulterhöhe sind.\n2. Setz dich mit dem Gesicht zum Polster, Brust daran, und greif die Griffe mit gestreckten Armen vor dir.\n3. Öffne die Arme in einem weiten Bogen zur Seite, bis sie in einer Linie mit dem Körper sind.\n4. Halte kurz inne und führ sie dann langsam nach vorne zurück.
exg_reverse_pec_deck_tips = • Halte die Ellbogen leicht gebeugt und drück die Griffe weit nach außen, nicht nach hinten.\n• Zieh die Schultern nicht hoch; lass die hinteren Schultern arbeiten.
exg_machine_row_steps = 1. Stell Sitz und Brustpolster so ein, dass die Griffe auf mittlerer Brusthöhe sind und du sie mit gestreckten Armen erreichst.\n2. Setz dich mit der Brust ans Polster und greif die Griffe.\n3. Zieh die Griffe zum Oberkörper, Ellbogen nach hinten, und drück die Schulterblätter zusammen.\n4. Lass langsam zurück, bis die Arme gestreckt sind und du eine Dehnung im Rücken spürst.
exg_machine_row_tips = • Lass die Brust am Polster; lehn dich nicht zurück, um das Gewicht zu bewegen.\n• Zieh mit den Ellbogen und halte die Schultern unten, nicht hochgezogen.
exg_pullover_machine_steps = 1. Stell den Sitz so ein, dass deine Schultern auf Höhe des Drehpunkts sind, und schließ den Gurt, falls vorhanden.\n2. Leg die Ellbogen an die Polster, Arme über dem Kopf, und halte die Stange locker.\n3. Führ die Arme im Bogen nach unten und vorne, bis die Stange am Bauch ist.\n4. Führ sie langsam über den Kopf zurück, bis du eine Dehnung im Latissimus spürst.
exg_pullover_machine_tips = • Drück mit den Ellbogen, nicht mit den Händen, damit der Rücken arbeitet.\n• Halte den unteren Rücken am Polster; mach unten keinen Rundrücken.
exg_leg_press_machine_steps = 1. Setz dich mit Rücken und Hüfte flach ans Polster, Füße schulterbreit in der Mitte der Plattform.\n2. Drück die Plattform leicht hoch und löse die Sicherung.\n3. Senk die Plattform ab, indem du die Knie auf etwa 90 Grad beugst, ohne dass die Hüfte vom Sitz abhebt.\n4. Drück mit dem ganzen Fuß wieder hoch und stopp kurz bevor die Knie durchgestreckt sind.\n5. Sichere die Plattform wieder, wenn du den Satz beendet hast.
exg_leg_press_machine_tips = • Halte die Knie in einer Linie mit den Zehen; lass sie nicht nach innen fallen.\n• Geh nicht so tief, dass sich der untere Rücken rundet und vom Polster abhebt.\n• Streck die Knie oben nie ganz durch.
exg_horizontal_leg_press_steps = 1. Stell die Rückenlehne so ein, dass deine Knie etwa 90 Grad gebeugt sind, die Füße auf der Plattform.\n2. Setz dich mit dem Rücken ans Polster, Füße schulterbreit, und halte die seitlichen Griffe.\n3. Drück die Plattform weg (oder je nach Modell den Sitz nach hinten), bis die Beine fast gestreckt sind.\n4. Kehr langsam zum Start zurück, ohne dass die Gewichte aufsetzen.
exg_horizontal_leg_press_tips = • Drück über Fersen und Mittelfuß, Knie in einer Linie mit den Zehen.\n• Lass Hüfte und unteren Rücken auf dem Sitz; streck die Knie nicht durch.
exg_hack_squat_machine_steps = 1. Stell dich auf die Plattform, Rücken am Polster, Schultern unter den Polstern, Füße schulterbreit.\n2. Streck die Beine, um den Schlitten anzuheben, und löse die Sicherung.\n3. Beug die Knie und geh in die Hocke, Rücken am Polster, bis die Oberschenkel etwa parallel zur Plattform sind.\n4. Drück dich mit dem ganzen Fuß wieder hoch, ohne die Knie durchzustrecken.\n5. Sichere den Schlitten wieder, wenn du den Satz beendet hast.
exg_hack_squat_machine_tips = • Lass die Fersen unten und die Knie über den Zehen.\n• Füße höher auf der Plattform trainieren mehr das Gesäß, tiefer mehr den Quadrizeps.\n• Lass die Hüfte unten nicht vom Polster abheben.
exg_pendulum_squat_machine_steps = 1. Stell dich mit dem Rücken ans Polster, Schultern unter den Polstern, Füße etwa hüftbreit.\n2. Richte dich auf, um das Gewicht zu übernehmen, und löse den Sicherungshebel.\n3. Geh in die Hocke, während das Polster im Bogen mitschwingt; lass die Knie nach vorne wandern, so tief du kontrollieren kannst.\n4. Drück dich mit dem ganzen Fuß hoch, ohne die Knie durchzustrecken; sichere die Maschine am Ende wieder.
exg_pendulum_squat_machine_tips = • Lass die Fersen unten und den Rücken die ganze Zeit fest am Polster.\n• Senk dich langsam ab und federe unten nicht.

Senk langsam ab, ohne dass die Gewichte aufsetzen.
exg_leg_extension_machine_tips = • Lass Hüfte und Rücken am Sitz; reiß das Gewicht nicht mit Schwung hoch.\n• Beweg dich gleichmäßig, ohne zu treten.
exg_lying_leg_curl_machine_steps = 1. Leg dich auf den Bauch, die Knie knapp über die Bankkante hinaus, auf Höhe des Drehpunkts.\n2. Leg die Rolle hinten auf die Unterschenkel, knapp über den Fersen, und halte die Griffe.\n3. Zieh die Fersen so weit wie möglich zum Gesäß.\n4. Senk langsam ab, bis die Beine fast gestreckt sind.
exg_lying_leg_curl_machine_tips = • Drück die Hüfte in die Bank; lass sie beim Beugen nicht hochkommen.\n• Reiß nicht am Gewicht; kontrolliere den Weg nach unten.

exg_seated_leg_curl_machine_tips = • Stell das Oberschenkelpolster fest ein, damit die Hüfte nicht abhebt.\n• Halte den Rücken an der Lehne und kontrolliere den Rückweg; lass das Gewicht nicht zurückschnellen.

exg_glute_kickback_machine_tips = • Die Bewegung kommt aus der Hüfte, nicht aus dem unteren Rücken: kein Hohlkreuz, um höher zu kommen.\n• Halte die Hüfte waagerecht; dreh dich nicht zur Seite.
exg_hip_thrust_machine_steps = 1. Setz dich mit dem oberen Rücken ans Polster und leg den Gurt oder das Polster über die Hüfte.\n2. exg_hip_thrust_machine_tips = • Zieh das Kinn leicht ein und halte die Rippen unten; mach oben kein Hohlkreuz.\n• Spürst du es vor allem in den Beinbeugern, stell die Füße etwas näher heran.

exg_back_extension_machine_tips = • Beweg dich gleichmäßig und stopp bei leichter Rücklage; erzwing die Streckung nicht.\n• Nimm ein moderates Gewicht: Das ist eine Kontrollübung, kein Maximalversuch.

exg_rotary_torso_machine_steps = 1. Stell den Startwinkel auf eine Seite ein und setz dich je nach Modell mit den Knien oder Beinen an die Polster.\n2. exg_rotary_torso_machine_tips = • Dreh aus der Taille und bleib langsam; kein Schwung.\n• Nimm wenig Gewicht und einen Bewegungsumfang, den du kontrollierst.

Greif die Stange etwas weiter als schulterbreit, Handflächen nach vorne, und setz dich.\n3. Zieh die Stange zur oberen Brust, Ellbogen nach unten und hinten.\n4. Lass ihn langsam hoch, bis die Arme gestreckt sind und der Latissimus gedehnt ist.
exg_close_grip_lat_pulldown_tips = • Halte die Brust oben und lehn dich nur leicht zurück.\n• Zieh mit den Ellbogen; reiß nicht mit den Armen und schwing nicht.
